# 1. Compilation release pour le web
cargo build --release --target wasm32-unknown-unknown

# 2. Génération du JS et du .wasm pour le navigateur
wasm-bindgen --no-typescript --out-name bevy_game --out-dir wasm --target web \
  target/wasm32-unknown-unknown/release/alchimie-des-formes.wasm

# 3. Réduction de la taille
wasm-opt -Oz --enable-bulk-memory --enable-nontrapping-float-to-int \
  --enable-sign-ext --enable-mutable-globals --enable-simd \
  -o wasm/bevy_game_bg.wasm wasm/bevy_game_bg.wasm

# 4. Copie des assets
rm -rf wasm/assets && cp -r assets wasm/

# 5. Archive pour itch.io
rm -f alchimie-web.zip && (cd wasm && zip -qr ../alchimie-web.zip . -x '.DS_Store')

Test local
cd wasm && python3 -m http.server 8000
